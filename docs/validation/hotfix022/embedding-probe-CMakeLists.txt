cmake_minimum_required(VERSION 3.14)
project(embedding_probe C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(/workspace/local-ai-workspace/third_party/llama.cpp llama)
add_executable(probe main.cpp)
target_include_directories(probe PRIVATE /workspace/local-ai-workspace/llama-runtime/src/main/cpp)
add_dependencies(probe ggml-cpu)
target_link_directories(probe PRIVATE "$<TARGET_FILE_DIR:ggml-cpu>")
target_link_libraries(probe llama "-l:$<TARGET_FILE_NAME:ggml-cpu>")

add_library(embedding-jni-host SHARED /workspace/local-ai-workspace/llama-runtime/src/main/cpp/local_embedding_jni.cpp)
target_include_directories(embedding-jni-host PRIVATE /workspace/local-ai-workspace/llama-runtime/src/main/cpp /workspace/.toolchain/jdk-17/include /workspace/.toolchain/jdk-17/include/linux)
add_dependencies(embedding-jni-host ggml-cpu)
target_link_directories(embedding-jni-host PRIVATE "$<TARGET_FILE_DIR:ggml-cpu>")
target_link_libraries(embedding-jni-host llama "-l:$<TARGET_FILE_NAME:ggml-cpu>")
